Sessão Privada Ativa
Script SQL de Migração — Onda 2 (Sincronização Total + RLS)

Cole e execute no SQL Editor do seu painel Supabase. É seguro rodar mais de uma vez (idempotente) e mantém seus lançamentos antigos acessíveis após o login. 

-- 1. Colunas de sincronizacao total (Onda 2)
ALTER TABLE transacoes ADD COLUMN IF NOT EXISTS bloco_orcamento text;
ALTER TABLE transacoes ADD COLUMN IF NOT EXISTS status_pagamento text DEFAULT 'pago';
ALTER TABLE reserva_movimentacoes ADD COLUMN IF NOT EXISTS afeta_caixa boolean DEFAULT true;
ALTER TABLE configuracoes ADD COLUMN IF NOT EXISTS extras_config jsonb DEFAULT '{}'::jsonb;

-- 2. Coluna user_id + RLS privado por usuario autenticado
ALTER TABLE transacoes ADD COLUMN IF NOT EXISTS user_id uuid DEFAULT auth.uid();
ALTER TABLE categorias ADD COLUMN IF NOT EXISTS user_id uuid DEFAULT auth.uid();
ALTER TABLE configuracoes ADD COLUMN IF NOT EXISTS user_id uuid DEFAULT auth.uid();
ALTER TABLE investimentos ADD COLUMN IF NOT EXISTS user_id uuid DEFAULT auth.uid();
ALTER TABLE reserva_movimentacoes ADD COLUMN IF NOT EXISTS user_id uuid DEFAULT auth.uid();
ALTER TABLE sonhos ADD COLUMN IF NOT EXISTS user_id uuid DEFAULT auth.uid();

DO $$
DECLARE tbl text;
BEGIN
  FOREACH tbl IN ARRAY ARRAY['transacoes','categorias','configuracoes','investimentos','reserva_movimentacoes','sonhos']
  LOOP
    EXECUTE format('ALTER TABLE %I ENABLE ROW LEVEL SECURITY;', tbl);
    EXECUTE format('DROP POLICY IF EXISTS "mp_owner_access" ON %I;', tbl);
    EXECUTE format(
      'CREATE POLICY "mp_owner_access" ON %I FOR ALL USING (auth.uid() IS NOT NULL AND (user_id = auth.uid() OR user_id IS NULL)) WITH CHECK (auth.uid() IS NOT NULL AND (user_id = auth.uid() OR user_id IS NULL));',
      tbl
    );
  END LOOP;
END $$;

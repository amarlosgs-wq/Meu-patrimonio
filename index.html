Script SQL das Tabelas (Supabase SQL Editor) 
CREATE TABLE IF NOT EXISTS transacoes (id UUID PRIMARY KEY DEFAULT gen_random_uuid(), tipo TEXT, valor NUMERIC, descricao TEXT, categoria TEXT, data DATE, recorrencia TEXT, natureza TEXT, essencialidade TEXT, forma_pagamento TEXT);
CREATE TABLE IF NOT EXISTS categorias (id UUID PRIMARY KEY DEFAULT gen_random_uuid(), nome TEXT, tipo TEXT);
CREATE TABLE IF NOT EXISTS configuracoes (id INT PRIMARY KEY DEFAULT 1, perc_economia NUMERIC, perc_entretenimento NUMERIC, perc_doacoes NUMERIC, perc_demais NUMERIC, meses_meta_reserva INT, custo_fixo_estimado NUMERIC);
CREATE TABLE IF NOT EXISTS investimentos (id UUID PRIMARY KEY DEFAULT gen_random_uuid(), nome TEXT, classe TEXT, instituicao TEXT, valor_aplicado NUMERIC, saldo_atual NUMERIC);
CREATE TABLE IF NOT EXISTS reserva_movimentacoes (id UUID PRIMARY KEY DEFAULT gen_random_uuid(), tipo TEXT, valor NUMERIC, onde_guardado TEXT, data DATE);
CREATE TABLE IF NOT EXISTS sonhos (id UUID PRIMARY KEY DEFAULT gen_random_uuid(), titulo TEXT, valor_alvo NUMERIC, valor_atual NUMERIC, data_limite DATE, prioridade TEXT);
